## 4-1　Secret/Podのデプロイ
# Namespace作成
$ kubectl create secret namespace k8s-defense

# Secret作成
$ kubectl create secret generic api-key-secret  -n k8s-defense \
  --from-literal=API_KEY=1234-SECRET-5678

# Podのデプロイ
$ kubectl apply -f secret-test-pod.yaml

# PodへのSecret情報反映確認（環境変数、ボリュームマウント）
$ kubectl exec -n k8s-defense secret-test-pod -- env | grep API_KEY
$ kubectl exec -n k8s-defense secret-test-pod -- cat /mnt/secrets/API_KEY

# Secret情報の暗号化確認
$ kubectl get secret api-key-secret -n k8s-defense -o yaml

# APIキーの復号化（base64）
$ echo “<暗号化されたキー>" | base64 --decode

## 4-4 AuthorizationPolicyによる認可

# 経路A（正規）：reviews v2 → ratings 通信（HTTP 200 OK）
$ kubectl exec -n book-info $(kubectl get pod -n book-info \
  -l app=reviews,version=v2 -o jsonpath='{.items[0].metadata.name}')  \
  -c reviews -- curl -s -o /dev/null -w \
  "HTTP %{http_code}\n" http://ratings:9080/ratings/0

# 経路B（他）：reviews v2 → details 通信（HTTP 200 OK）
$ kubectl exec -n book-info $(kubectl get pod -n book-info \
  -l app=reviews,version=v2 -o jsonpath='{.items[0].metadata.name}') \
  -c reviews -- curl -s -o /dev/null \
  -w "HTTP %{http_code}\n" http://details:9080/details/0

# 経路C（正規）：productpage → details 通信（200OK）
$ kubectl exec -n book-info $(kubectl get pod -n book-info \
  -l app=productpage -o jsonpath='{.items[0].metadata.name}') \
  -c productpage -- python -c "import requests; print(requests.get('http://details:9080/details/0').status_code)"

# 経路D（他）：Details → reviews への通信確認（200OK）
$ kubectl exec -n book-info $(kubectl get pod -n book-info \
  -l app=details -o jsonpath='{.items[0].metadata.name}') \
  -c details -- ruby -rnet/http -e "uri=URI('http://reviews:9080/reviews/1'); res=Net::HTTP.get_response(uri); puts res.code"

# AuthorizationPolicyの適用
$ kubectl apply –f autho-deny-all.yaml
$ kubectl apply –f autho-productpage.yaml
$ kubectl apply –f autho-reviews.yaml
$ kubectl apply –f autho-ratings.yaml
$ kubectl apply –f autho-details.yaml

# AuthorizationPolicyの設定確認
$ kubectl get authorizationpolicy -n book-info
